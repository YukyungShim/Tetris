-- 1. 테트리스 랭킹 테이블 생성
CREATE TABLE IF NOT EXISTS public.tetris_leaderboard (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- 닉네임 (최대 12글자, 중복 방지 및 유저별 최고기록 관리)
    name VARCHAR(12) NOT NULL UNIQUE, 
    -- 테트리스 최고 점수 (0 ~ 9,999,999)
    score INTEGER NOT NULL CHECK (score >= 0 AND score <= 9999999),
    -- 도달한 최고 라운드 (기본값 1)
    round_reached INTEGER NOT NULL DEFAULT 1 CHECK (round_reached >= 1),
    -- 클리어한 총 줄 수 (기본값 0)
    lines_cleared INTEGER NOT NULL DEFAULT 0 CHECK (lines_cleared >= 0),
    -- 기록 달성 일시
    played_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. 랭킹 조회를 위한 고속 인덱스 생성
-- 고득점순(DESC), 동점일 경우 먼저 달성한 시간순(ASC)으로 정렬
CREATE INDEX IF NOT EXISTS idx_tetris_score_desc 
ON public.tetris_leaderboard (score DESC, played_at ASC);

-- 3. Row Level Security (RLS) 활성화
ALTER TABLE public.tetris_leaderboard ENABLE ROW LEVEL SECURITY;

-- 4. 익명 읽기(SELECT) 허용 정책 추가 (누구나 랭킹을 조회할 수 있음)
CREATE POLICY "Allow public read access" 
ON public.tetris_leaderboard
FOR SELECT 
USING (true);

-- 5. 익명 쓰기(INSERT/UPDATE) 허용 정책 추가 (프론트/서버리스 API에서 랭킹 등록 및 갱신)
CREATE POLICY "Allow public insert and update" 
ON public.tetris_leaderboard
FOR ALL 
USING (true)
WITH CHECK (true);
